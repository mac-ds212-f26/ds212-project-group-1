Visualization 1: When are goalies pulled?

This visualization shows how many candidate goalie pulls happen at different times in the third period.

Code 
# Filter third-period goalie pulls
pull_timing <- pulls %>%
  filter(
    period == 3,
    !is.na(pull_time_remaining_sec),
    between(pull_time_remaining_sec, 0, 600)
  ) %>%
  mutate(minutes_remaining = pull_time_remaining_sec / 60)

ggplot(pull_timing, aes(x = minutes_remaining)) +
  geom_histogram(
    binwidth = 1,
    fill = "steelblue",
    color = "white"
  ) +
  scale_x_continuous(breaks = 0:10) +
  labs(
    title = "When Do Teams Pull Their Goalies?",
    subtitle = "PWHL candidate goalie pulls in the final 10 minutes",
    x = "Minutes remaining in the third period",
    y = "Number of goalie pulls"
  ) +
  theme_minimal()
